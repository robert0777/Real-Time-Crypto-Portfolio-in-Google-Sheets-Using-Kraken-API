function importKrakenData() {
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  var sheetName = "BNB/USD";
  var sheet = spreadsheet.getSheetByName(sheetName);
  
  // Create the sheet if it doesn't exist
  if (!sheet) {
    sheet = spreadsheet.insertSheet(sheetName);
  }
  
  // Set the new sheet as the active sheet
  spreadsheet.setActiveSheet(sheet);
  
  // Clear any existing content in the sheet
  sheet.clear();
  
  // Kraken API for Binance OHLC data (USD, 1-day interval)
  var url = "https://api.kraken.com/0/public/OHLC?pair=BNBUSD&interval=1440";
  
  try {
    Utilities.sleep(2000); // Delay to prevent rate limiting

    var response = UrlFetchApp.fetch(url, { muteHttpExceptions: true });

    if (response.getResponseCode() !== 200) {
      Logger.log("Response Code: " + response.getResponseCode());
      Logger.log("Response: " + response.getContentText());
      throw new Error("Failed to fetch data from Kraken API");
    }

    var json = JSON.parse(response.getContentText());

    // Extract OHLC data
    var data = json.result.BNBUSD; // BNB/USD pair on Kraken

    // Prepare headers
    var headers = [["Timestamp", "Open", "High", "Low", "Close"]];
    sheet.getRange(1, 1, 1, 5).setValues(headers);

    // Format data - ensure precision is maintained
    var formattedData = data.map(function(row) {
      return [
        new Date(row[0] * 1000), // Convert Unix timestamp to date
        parseFloat(row[1]), // Open
        parseFloat(row[2]), // High
        parseFloat(row[3]), // Low
        parseFloat(row[4])  // Close
      ];
    });

    // Write data to sheet
    if (formattedData.length > 0) {
      sheet.getRange(2, 1, formattedData.length, 5).setValues(formattedData);
      
      // Format timestamp column
      sheet.getRange(2, 1, formattedData.length, 1).setNumberFormat("yyyy-mm-dd hh:mm:ss");
      
      // Format price columns with more decimal places
      // Use #,##0.0000 for 4 decimal places, or increase as needed
      sheet.getRange(2, 2, formattedData.length, 4).setNumberFormat("$#,##0.000000");
      
      // Alternative: Use scientific notation if you need even more precision
      // sheet.getRange(2, 2, formattedData.length, 4).setNumberFormat("$0.00000E+00");
      
      // Or use general number format to show all available digits
      // sheet.getRange(2, 2, formattedData.length, 4).setNumberFormat("General");
    }

  } catch (error) {
    Logger.log("Error: " + error.toString());
    throw new Error("Failed to import data from Kraken API");
  }
}
