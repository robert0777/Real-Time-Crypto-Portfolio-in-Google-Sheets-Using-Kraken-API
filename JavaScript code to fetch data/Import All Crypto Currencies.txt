function importCryptoData(spreadsheet, crypto) {
  var sheetName = crypto.name;
  var sheet = spreadsheet.getSheetByName(sheetName);
  
  // Create the sheet if it doesn't exist
  if (!sheet) {
    sheet = spreadsheet.insertSheet(sheetName);
  }
  
  // Clear any existing content in the sheet
  sheet.clear();
  
  // Kraken API URL
  var url = "https://api.kraken.com/0/public/OHLC?pair=" + crypto.pair + "&interval=1440";
  
  try {
    var response = UrlFetchApp.fetch(url, { 
      muteHttpExceptions: true,
      headers: {
        'User-Agent': 'Google Apps Script'
      }
    });

    if (response.getResponseCode() !== 200) {
      Logger.log("Response Code for " + crypto.name + ": " + response.getResponseCode());
      Logger.log("Response: " + response.getContentText());
      throw new Error("Failed to fetch data from Kraken API for " + crypto.name);
    }

    var json = JSON.parse(response.getContentText());
    
    // Check if the API returned an error
    if (json.error && json.error.length > 0) {
      throw new Error("Kraken API Error for " + crypto.name + ": " + json.error.join(", "));
    }

    // Extract OHLC data using the correct result key
    var data = json.result[crypto.resultKey];
    
    if (!data || data.length === 0) {
      throw new Error("No data returned for " + crypto.name);
    }

    // Prepare headers with crypto name
    var headers = [["Timestamp", "Open (" + crypto.name + ")", "High (" + crypto.name + ")", "Low (" + crypto.name + ")", "Close (" + crypto.name + ")", "Volume"]];
    sheet.getRange(1, 1, 1, 6).setValues(headers);
    
    // Style headers
    var headerRange = sheet.getRange(1, 1, 1, 6);
    headerRange.setFontWeight("bold");
    headerRange.setBackground("#4285f4");
    headerRange.setFontColor("white");

    // Format data
    var formattedData = data.map(function(row) {
      return [
        new Date(row[0] * 1000), // Convert Unix timestamp to date
        parseFloat(row[1]), // Open
        parseFloat(row[2]), // High
        parseFloat(row[3]), // Low
        parseFloat(row[4]), // Close
        parseFloat(row[6])  // Volume
      ];
    });

    // Write data to sheet
    if (formattedData.length > 0) {
      sheet.getRange(2, 1, formattedData.length, 6).setValues(formattedData);
      
      // Format timestamp column
      sheet.getRange(2, 1, formattedData.length, 1).setNumberFormat("yyyy-mm-dd hh:mm:ss");
      
      // Format price columns (Open, High, Low, Close)
      sheet.getRange(2, 2, formattedData.length, 4).setNumberFormat("$#,##0.000000");
      
      // Format volume column
      sheet.getRange(2, 6, formattedData.length, 1).setNumberFormat("#,##0.00");
      
      // Auto-resize columns
      sheet.autoResizeColumns(1, 6);
    }

  } catch (error) {
    Logger.log("Error importing " + crypto.name + ": " + error.toString());
    throw error;
  }
}

// Function to create a summary sheet with latest prices
function createSummarySheet() {
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  var summarySheetName = "Crypto Summary";
  var summarySheet = spreadsheet.getSheetByName(summarySheetName);
  
  if (!summarySheet) {
    summarySheet = spreadsheet.insertSheet(summarySheetName);
  }
  
  summarySheet.clear();
  
  // Headers for summary
  var headers = [["Cryptocurrency", "Latest Price", "24h Change", "Last Updated"]];
  summarySheet.getRange(1, 1, 1, 4).setValues(headers);
  
  // Style headers
  var headerRange = summarySheet.getRange(1, 1, 1, 4);
  headerRange.setFontWeight("bold");
  headerRange.setBackground("#34a853");
  headerRange.setFontColor("white");
  
  var cryptos = ["BTC/USD", "ETH/USD", "XRP/USD", "DOGE/USD", "ADA/USD", "LTC/USD", "SHIB/USD", "USDT/USD", "BNB/USD", "DAI/USD", "PYUSD/USD", "TUSD/USD"];
  var summaryData = [];
  
  cryptos.forEach(function(cryptoName) {
    var sheet = spreadsheet.getSheetByName(cryptoName);
    if (sheet && sheet.getLastRow() > 1) {
      var lastRow = sheet.getLastRow();
      var latestPrice = sheet.getRange(lastRow, 5).getValue(); // Close price
      var previousPrice = lastRow > 2 ? sheet.getRange(lastRow - 1, 5).getValue() : latestPrice;
      var change = ((latestPrice - previousPrice) / previousPrice * 100).toFixed(2);
      var timestamp = sheet.getRange(lastRow, 1).getValue();
      
      summaryData.push([cryptoName, latestPrice, change + "%", timestamp]);
    }
  });
  
  if (summaryData.length > 0) {
    summarySheet.getRange(2, 1, summaryData.length, 4).setValues(summaryData);
    summarySheet.getRange(2, 2, summaryData.length, 1).setNumberFormat("$#,##0.000000");
    summarySheet.getRange(2, 4, summaryData.length, 1).setNumberFormat("yyyy-mm-dd hh:mm:ss");
    summarySheet.autoResizeColumns(1, 4);
  }
}
